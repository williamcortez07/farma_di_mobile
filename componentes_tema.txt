class AppTheme {
  AppTheme._();

  static ThemeData get light {
    const colorScheme = ColorScheme(
      brightness: Brightness.light,
      primary: AppColors.primary,
      onPrimary: Colors.white,
      secondary: AppColors.accent,
      onSecondary: Colors.white,
      error: Color(0xFFBA1A1A),
      onError: Colors.white,
      surface: Colors.white,
      onSurface: AppColors.textPrimary,
    );

    return ThemeData(
      useMaterial3: true,
      brightness: Brightness.light,
      colorScheme: colorScheme,
      scaffoldBackgroundColor: AppColors.background,
      fontFamily: 'Roboto',
      appBarTheme: const AppBarTheme(
        backgroundColor: AppColors.background,
        foregroundColor: AppColors.textPrimary,
        elevation: 0,
        centerTitle: false,
      ),
      cardTheme: const CardThemeData(
        color: AppColors.surface,
        elevation: 0,
        margin: EdgeInsets.zero,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.all(Radius.circular(18)),
        ),
      ),
      inputDecorationTheme: InputDecorationTheme(
        filled: true,
        fillColor: AppColors.surface,
        hintStyle: const TextStyle(color: AppColors.textSecondary),
        border: OutlineInputBorder(
          borderRadius: BorderRadius.circular(12),
          borderSide: const BorderSide(color: AppColors.border),
        ),
        enabledBorder: OutlineInputBorder(
          borderRadius: BorderRadius.circular(12),
          borderSide: const BorderSide(color: AppColors.border),
        ),
        focusedBorder: OutlineInputBorder(
          borderRadius: BorderRadius.circular(12),
          borderSide: const BorderSide(color: AppColors.primary, width: 1.5),
        ),
      ),
      dividerColor: AppColors.border,
      switchTheme: SwitchThemeData(
        thumbColor: const WidgetStatePropertyAll(Colors.white),
        trackColor: WidgetStateProperty.resolveWith((states) {
          if (states.contains(WidgetState.selected)) {
            return AppColors.primary;
          }
          return const Color(0xFFE2E5EA);
        }),
        trackOutlineColor: const WidgetStatePropertyAll(Colors.transparent),
      ),
    );
  }

    const surface = Color(0xFF1D2732);
    const primary = Color(0xFF83B2FF);

    return ThemeData(
      useMaterial3: true,
      brightness: Brightness.dark,

/// KPI 3 â€“ ParticipaciÃ³n de ventas por categorÃ­a (%).
///
/// Muestra un [PieChart] con la distribuciÃ³n porcentual de ingresos por
/// categorÃ­a, acompaÃ±ado de una leyenda compacta.
class CategoryShareChart extends StatefulWidget {
  const CategoryShareChart({
    super.key,
    required this.data,
  });

  final List<CategoryShareModel> data;

  @override
  State<CategoryShareChart> createState() => _CategoryShareChartState();
}

class _CategoryShareChartState extends State<CategoryShareChart> {
  int _touchedIndex = -1;

  static const List<Color> _palette = [
    AppColors.primaryDark,
    AppColors.chartBlue,
    AppColors.positiveGreen,
    Color(0xFF8B7FE0),
    AppColors.warningOrange,
    Color(0xFFE0645F),
    Color(0xFFE0A438),
    AppColors.neutralGray,
  ];

  @override
  Widget build(BuildContext context) {
    if (widget.data.isEmpty) return const _EmptyState();

    return Column(
      children: [
        SizedBox(
          height: 200,
          child: PieChart(
            PieChartData(
              pieTouchData: PieTouchData(
                touchCallback: (event, response) {
                  setState(() {
                    if (!event.isInterestedForInteractions ||
                        response == null ||
                        response.touchedSection == null) {
                      _touchedIndex = -1;
                      return;
                    }
                    _touchedIndex =
                        response.touchedSection!.touchedSectionIndex;
                  });
                },
              ),
              startDegreeOffset: -90,
              sectionsSpace: 2,
              centerSpaceRadius: 48,
              sections: List.generate(widget.data.length, (i) {
                final isTouched = i == _touchedIndex;
                final item = widget.data[i];
                return PieChartSectionData(
                  color: _palette[i % _palette.length],
                  value: item.sharePercentage,
                  radius: isTouched ? 70 : 58,
                  title: isTouched
                      ? '${item.sharePercentage.toStringAsFixed(1)}%'
                      : '',
                  titleStyle: const TextStyle(
                    fontSize: 11,
                    fontWeight: FontWeight.w700,
                    color: Colors.white,
                  ),
                );
              }),
            ),
          ),
        ),
        const SizedBox(height: 16),
        _buildLegend(),
      ],
    );
  }

  Widget _buildLegend() {
    return GridView.builder(
      shrinkWrap: true,
      physics: const NeverScrollableScrollPhysics(),
      itemCount: widget.data.length,
      gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
        crossAxisCount: 2,
        mainAxisExtent: 24,
        crossAxisSpacing: 8,
        mainAxisSpacing: 4,
      ),
      itemBuilder: (context, i) {
        final item = widget.data[i];
        final color = _palette[i % _palette.length];
        final isHighlighted = i == _touchedIndex;
        return GestureDetector(
          onTap: () => setState(
              () => _touchedIndex = _touchedIndex == i ? -1 : i),
          child: Row(
            children: [
              Container(
                width: 8,
                height: 8,
                decoration: BoxDecoration(
                  color: color,
                  shape: BoxShape.circle,
                ),
              ),
              const SizedBox(width: 6),
              Expanded(
                child: Text(
                  item.categoryName,
                  overflow: TextOverflow.ellipsis,
                  style: TextStyle(
                    fontSize: 10,
                    fontWeight: isHighlighted
                        ? FontWeight.w700
                        : FontWeight.w400,
                    color: isHighlighted
                        ? AppColors.primaryDark
                        : AppColors.mutedText,
                  ),
                ),
              ),
              Text(
                '${item.sharePercentage.toStringAsFixed(1)}%',
                style: TextStyle(
                  fontSize: 10,
                  fontWeight: FontWeight.w700,
                  color: isHighlighted ? color : AppColors.mutedText,
                ),
              ),
            ],
          ),
        );
      },
    );
  }
}

/// KPI 2 â€“ Margen bruto de ventas (%).
///
/// Muestra un [HorizontalBarChart] (barras horizontales) con el porcentaje
/// de margen bruto por producto.  Las barras se presentan de mayor a menor
/// margen segÃºn el ordenamiento aplicado en [SalesMetricsService].
class GrossMarginChart extends StatelessWidget {
  const GrossMarginChart({super.key, required this.data});

  final List<GrossMarginModel> data;
  static const double _barHeight = 14;
  static const double _rowSpacing = 28;

  @override
  Widget build(BuildContext context) {
    if (data.isEmpty) return const _EmptyState();

    final maxValue = data
        .map((e) => e.grossMargin)
        .reduce((a, b) => a > b ? a : b);

    return SizedBox(
      height: data.length * _rowSpacing + 20,
      child: BarChart(
        BarChartData(
          alignment: BarChartAlignment.center,
          maxY: data.length.toDouble(),
          minY: 0,
          rotationQuarterTurns: 1, // hace las barras horizontales
          gridData: const FlGridData(show: false),
          borderData: FlBorderData(show: false),
          barTouchData: BarTouchData(
            touchTooltipData: BarTouchTooltipData(
              getTooltipColor: (_) => AppColors.primaryDark,
              getTooltipItem: (group, groupIndex, rod, rodIndex) {
                final item = data[group.x];
                return BarTooltipItem(
                  '${item.productName}\n'
                  'Margen: C\$${item.grossMargin.toStringAsFixed(0)}\n'
                  '${item.grossMarginPercentage.toStringAsFixed(2)}%',
                  const TextStyle(
                    color: Colors.white,
                    fontSize: 11,
                    fontWeight: FontWeight.w600,
                  ),
                );
              },
            ),
          ),
          titlesData: FlTitlesData(
            topTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),
            rightTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),
            // Eje izquierdo â†’ nombres de productos (aparece abajo por la rotaciÃ³n)
            leftTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 90,
                getTitlesWidget: (value, meta) {
                  final index = value.toInt();
                  if (index < 0 || index >= data.length) {
                    return const SizedBox.shrink();
                  }
                  return Padding(
                    padding: const EdgeInsets.only(right: 6),
                    child: Text(
                      data[index].productName,
                      textAlign: TextAlign.right,
                      style: const TextStyle(
                        fontSize: 10,
                        color: AppColors.primaryDark,
                      ),
                    ),
                  );
                },
              ),
            ),
            // Eje superior â†’ valor del margen (aparece a la derecha por la rotaciÃ³n)
            bottomTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 30,
                getTitlesWidget: (value, meta) {
                  final index = value.toInt();
                  if (index < 0 || index >= data.length) {
                    return const SizedBox.shrink();
                  }
                  return Padding(
                    padding: const EdgeInsets.only(left: 4),
                    child: Text(
                      'C\$${data[index].grossMargin.toStringAsFixed(0)}',
                      style: const TextStyle(
                        fontSize: 9,
                        color: AppColors.mutedText,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                  );
                },
              ),
            ),
          ),
          barGroups: List.generate(data.length, (i) {
            final normalizedValue =
                (data[i].grossMargin / maxValue) * (data.length - 0.5);
            return BarChartGroupData(
              x: i,
              barRods: [
                BarChartRodData(
                  toY: normalizedValue,
                  color: _barColor(i),
                  width: _barHeight,
                  borderRadius: const BorderRadius.only(
                    topLeft: Radius.circular(4),
                    topRight: Radius.circular(4),
                  ),
                  backDrawRodData: BackgroundBarChartRodData(
                    show: true,
                    toY: data.length.toDouble(),
                    color: AppColors.trackBackground,
                  ),
                ),
              ],
            );
          }),
        ),
      ),
    );
  }

  Color _barColor(int index) {
    const colors = [
      AppColors.primaryDark,
      AppColors.chartBlue,
      AppColors.positiveGreen,
      Color(0xFF8B7FE0),
      AppColors.warningOrange,
      Color(0xFFE0645F),
      AppColors.chartBlue,
      AppColors.positiveGreen,
    ];
    return colors[index % colors.length];
  }
}

/// KPI 1 â€“ Ventas por periodo (%).
///
/// Muestra un [BarChart] con el monto de ventas por periodo y una etiqueta
/// de variaciÃ³n porcentual sobre cada barra (excepto el primer periodo que
/// no tiene referencia anterior).
class SalesVariationChart extends StatelessWidget {
  const SalesVariationChart({super.key, required this.data});

  final List<SalesVariationModel> data;

  static final _currencyFmt = NumberFormat.compactCurrency(
    locale: 'es_NI',
    symbol: 'C\$',
    decimalDigits: 0,
  );

  @override
  Widget build(BuildContext context) {
    if (data.isEmpty) return const _EmptyState();

    // Rango del eje Y con un 20% de margen superior para las etiquetas.
    final maxY =
        data.map((e) => e.currentSales).reduce((a, b) => a > b ? a : b) * 1.25;

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        // Leyenda de variaciÃ³n porcentual
        _buildVariationLegend(),
        const SizedBox(height: 12),
        SizedBox(
          height: 200,
          child: BarChart(
            BarChartData(
              maxY: maxY,
              minY: 0,
              gridData: FlGridData(
                show: true,
                drawVerticalLine: false,
                horizontalInterval: maxY / 4,
                getDrawingHorizontalLine: (_) =>
                    FlLine(color: AppColors.trackBackground, strokeWidth: 1),
              ),
              borderData: FlBorderData(show: false),
              barTouchData: BarTouchData(
                touchTooltipData: BarTouchTooltipData(
                  getTooltipColor: (_) => AppColors.primaryDark,
                  getTooltipItem: (group, groupIndex, rod, rodIndex) {
                    final item = data[group.x];
                    return BarTooltipItem(
                      '${item.period}\n${_currencyFmt.format(item.currentSales)}',
                      const TextStyle(
                        color: Colors.white,
                        fontSize: 12,
                        fontWeight: FontWeight.w600,
                      ),
                    );
                  },
                ),
              ),
              titlesData: FlTitlesData(
                topTitles: AxisTitles(
                  sideTitles: SideTitles(
                    showTitles: true,
                    reservedSize: 28,
                    getTitlesWidget: (value, meta) {
                      final index = value.toInt();
                      if (index < 0 || index >= data.length) {
                        return const SizedBox.shrink();
                      }
                      final item = data[index];
                      if (item.variationPercentage == null) {
                        return const SizedBox.shrink();
                      }
                      final pct = item.variationPercentage!;
                      return Padding(
                        padding: const EdgeInsets.only(bottom: 4),
                        child: Text(
                          '${pct >= 0 ? '+' : ''}${pct.toStringAsFixed(1)}%',
                          style: TextStyle(
                            fontSize: 10,
                            fontWeight: FontWeight.w700,
                            color: pct >= 0
                                ? AppColors.positiveGreen
                                : Colors.redAccent,
                          ),
                        ),
                      );
                    },
                  ),
                ),
                leftTitles: const AxisTitles(
                  sideTitles: SideTitles(showTitles: false),
                ),
                rightTitles: const AxisTitles(
                  sideTitles: SideTitles(showTitles: false),
                ),
                bottomTitles: AxisTitles(
                  sideTitles: SideTitles(
                    showTitles: true,
                    reservedSize: 28,
                    getTitlesWidget: (value, meta) {
                      final index = value.toInt();
                      if (index < 0 || index >= data.length) {
                        return const SizedBox.shrink();
                      }
                      return Padding(
                        padding: const EdgeInsets.only(top: 8),
                        child: Text(
                          data[index].period,
                          style: const TextStyle(
                            fontSize: 11,
                            color: AppColors.mutedText,
                          ),
                        ),
                      );
                    },
                  ),
                ),
              ),
              barGroups: List.generate(data.length, (i) {
                return BarChartGroupData(
                  x: i,
                  barRods: [
                    BarChartRodData(
                      toY: data[i].currentSales,
                      color: AppColors.chartBlue,
                      width: 36,
                      borderRadius: const BorderRadius.vertical(
                        top: Radius.circular(6),
                      ),
                    ),
                  ],
                );
              }),
            ),
          ),
        ),
      ],
    );
  }

  Widget _buildVariationLegend() {
    final withVariation = data
        .where((e) => e.variationPercentage != null)
        .toList();
    if (withVariation.isEmpty) return const SizedBox.shrink();

    return Wrap(
      spacing: 16,
      runSpacing: 4,
      children: withVariation.map((e) {
        final pct = e.variationPercentage!;
        final isPositive = pct >= 0;

/// Tarjeta visual reutilizable para representar un registro de Log
class LogCard extends StatelessWidget {
  const LogCard({
    super.key,
    required this.log,
    this.onTap,
  });

  final SystemLogModel log;
  final VoidCallback? onTap;

  String _normalizedLevel() {
    final lvl = (log.level ?? '').toUpperCase();
    if (lvl.contains('ERROR')) return 'ERROR';
    if (lvl.contains('WARN')) return 'WARN';
    return 'INFO';
  }

  Color _getLevelColor() {
    switch (_normalizedLevel()) {
      case 'ERROR':
        return AppColors.red;
      case 'WARN':
        return AppColors.warning;
      default:
        return AppColors.primary;
    }
  }

  Color _getLevelBgColor() {
    switch (_normalizedLevel()) {
      case 'ERROR':
        return AppColors.redSoft;
      case 'WARN':
        return AppColors.warningSoft;
      default:
        return AppColors.primarySoft;
    }
  }

  String _formatDateTime(DateTime? dt) {
    if (dt == null) return 'N/D';
    final y = dt.year.toString();
    final m = dt.month.toString().padLeft(2, '0');
    final d = dt.day.toString().padLeft(2, '0');
    final h = dt.hour.toString().padLeft(2, '0');
    final min = dt.minute.toString().padLeft(2, '0');
    final s = dt.second.toString().padLeft(2, '0');
    return '$y-$m-$d $h:$min:$s';
  }

  void _handleOpenDetails(BuildContext context) {
    if (onTap != null) {
      onTap!();
    } else {
      LogDetailSheet.show(context, log);
    }
